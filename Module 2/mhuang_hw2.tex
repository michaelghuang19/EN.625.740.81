\section*{1.}
Let \(X\) have an exponential distribution
\[
  p(x|v) = 
  \begin{cases}
    v e^{-v x}, & x \geq 0 \\
    0, & \text{otherwise.}
  \end{cases}
\]

\subsection*{(a)} 
Sketch \(p(x|v)\) versus \(x\) for a fixed value of the parameter \(v\). \\ \\


\subsection*{(b)}
Sketch \(p(x|v)\) versus \(v\), \(v > 0\) for a fixed value of \(x\). \\ \\


\subsection*{(c)}
Suppose that \(n\) samples \(x_1, \dots, x_n\) are drawn independently according to \(p(x|v)\). Show that the maximum likelyhood estimate for \(v\) is given by 
\[
  \hat{v} = \frac{1}{\frac{1}{n} \sum_{k=1}^n x_k}
\]
